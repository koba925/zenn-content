最後は or パターンを作ります。Toil の処理系では以下のような形で使っているものです。

```py
        match expr:
            case None | bool() | int() | str(): return expr
```

Python と同様に、or パターンには `|` を使います。`or` の使いまわしも考えたんですが、後に `or` はマクロによって `if` に変換されてしまい、評価時には AST から消えてしまう運命なので断念しました。

であればまず `|` を導入する必要があります。`or` にしようかと考えたくらいなので優先順位や結合は `or` と同じにします。

```diff py
 class Scanner:
     ...

     def tokenize(self) -> list[Token]:
         while (c := self._current_char()) != "$EOF":
             self._start_pos = self._current_pos
             match c:
                 ...
                 case c if c in "=<>!:":
                     ...
-                case c if c in "+-*/%()[]{}.,;":
+                case c if c in "+-*/%|()[]{}.,;":
                     self._tokens.append(Ident(c)); self._advance()
                 case invalid:
                     assert False, f"Invalid character @ tokenize(): {invalid}"
         ...

 class Parser:
     ...

     def _and_or(self):
         return self._binary_left({
-            Ident("and"): Ident("and"), Ident("or"): Ident("or")
+            Ident("and"): Ident("and"), Ident("or"): Ident("or"),
+            Ident("|"): Ident("|")
         }, self._not)
```

`bind()` はこうなります。

```diff py
 class Environment:
     ...

     def bind(self, pattern: Expr, value: Value) -> bool:
         match pattern:
             ...
             case (Ident("tuple"), sub_pats):
                 return isinstance(value, tuple) and self._bind_seq(sub_pats, value)
+            case (Ident("|"), [left_pat, right_pat]):
+                return self.bind(left_pat, value) or self.bind(right_pat, value)
             case (Ident(typ), [val_pat]):
                 return type(value).__name__ == typ and self.bind(val_pat, value)
             ...
```

`(Ident("|"), [left_pat, right_pat])` という形の AST だったら、まず左側のパターンを値とマッチさせ、失敗したら右のパターンをマッチさせています。`or` はそういう動きをするのでしたね（「4.5 and・or」を参照）。

実行例です。

```py
    # Example

    print("Or Pattern:")

    print("Match:")

    print(toil.ast(r""" match x case bool(bn) | int(bn) then bn end """))
    # -> (match, [x, [((|, [(bool, [bn]), (int, [bn])]), bn)]])
```

`|` は演算子ですが、関数呼び出しと同じ形の AST に変換されます。

```py
    toil.walk(r"""
        def foo(x) do
            match x
                case 2 | 3 | 4 then ['small int', x]
                case bool(bn) | int(bn) then ['bool or int', bn]
                case [a, b] | tuple(a, b) then ['list or tuple', a, b]
                case _ then ['else', x]
            end
        end
    """)

    print(toil.walk(r""" foo(2) """)) # -> ['small int', 2]
    print(toil.walk(r""" foo(False) """)) # -> ['bool or int', False]
    print(toil.walk(r""" foo(5) """)) # -> ['bool or int', 5]
    print(toil.walk(r""" foo([2, 3]) """)) # -> ['list or tuple', 2, 3]
    print(toil.walk(r""" foo(tuple(4, 5)) """)) # -> ['list or tuple', 4, 5]
    print(toil.walk(r""" foo({}) """)) # -> ['else', {}]
```

`|` の左右には自由にパターンが置けます。3 つのパターンの or もできています。`bool(bn) | int(bn)` のように `|` の左右に同じ変数名 `bn` を書いているのは、どちらにマッチしても `then` の後ろで `bn` として値を受け取れるようにするためです。Python では左右で変数名が違うとエラーになりますが、Toil ではチェックしていません。いつもどおりですね！

```py
    print("Parameter match:")

    toil.walk(r""" def foo(bool(bn) | int(bn)) do bn end """)
    print(toil.walk(r""" foo(2) """)) # -> 2
    print(toil.walk(r""" foo(True) """)) # -> True
    # print(toil.walk(r""" foo([]) """)) # -> Pattern mismatch

    print("Destructuring bind:")

    print(toil.walk(r""" bool(bn) | int(bn) := 2; bn """)) # -> 2
    print(toil.walk(r""" bool(bn) | int(bn) := True; bn """)) # -> True
    # print(toil.walk(r""" bool(bn) | int(bn) := []; bn """)) # -> Pattern mismatch
```

やはりあまり使わなさそうですが関数呼び出しや分割代入でも or パターンが動きます。

```py
    print("Fibonacci:")

    toil.walk(r"""
        def fib_rec(n) do
            match n
                case 0 | 1 then n
                case _ then fib_rec(n - 1) + fib_rec(n - 2)
            end
        end
    """)
    print(toil.walk(r""" fib_rec(0) """)) # -> 0
    print(toil.walk(r""" fib_rec(1) """)) # -> 1
    print(toil.walk(r""" fib_rec(6) """)) # -> 8
```

フィボナッチの match 式もよりすっきり書けるようになりました。

まだ Python のパターンマッチ機能をすべて実装できたわけではありませんが、かなりの表現力を持ったパターンマッチが実現できました。
