以下のように書いてクラスっぽいものを作ることができました。

```
def Animal(name) do
    self := {};
    self._name = name;
    self.introduce = func self do print("I am", self._name) end;
    self.make_sound = func self do print("crying") end;
    self.rename = func self, name do self._name = name end;
    self
end
```

これを以下のように書けたらもうオブジェクト指向言語と言っても過言ではないのではないでしょうか（過言

```
defclass Animal(name) do
    self._name = name;
    defmethod introduce do print("I am", self._name) end;
    defmethod make_sound do print("crying") end;
    defmethod rename(name) do self._name = name end
end
```

そうです構文糖衣です。

`defclass` とか `defmethod` を見かけたら `def Animal` みたいな AST にしてやります。

どんな AST を出力すればいいんでしょうか。実際に AST を見てみましょう。

```py
    print(toil.ast(r"""
        def Animal(name) do
            self := {};
            self._name = name;
            self.introduce = func self do print("I am", self._name) end;
            self.make_sound = func self do print("crying") end;
            self.rename = func self, name do self._name = name end;
            self
        end
    """))
```

実行してみます。

```py
(define, [Animal, (func, [[name], (seq, [(define, [self, {}]), (assign, [(dot, [self, '_name']), name]), (assign, [(dot, [self, 'introduce']), (func, [[self], (print, ['I am', (dot, [self, '_name'])])])]), (assign, [(dot, [self, 'make_sound']), (func, [[self], (print, ['crying'])])]), (assign, [(dot, [self, 'rename']), (func, [[self, name], (assign, [(dot, [self, '_name']), name])])]), self])])])
```

ちょっとげげっとしますね。エディタに貼り付けて整形しました。

```py
(define, [Animal, (func, [[name],
    (seq, [
        (define, [self, {}]),
        (assign, [(dot, [self, '_name']), name]),
        (assign, [(dot, [self, 'introduce']), (func, [[self],
            (print, ['I am', (dot, [self, '_name'])])])]),
        (assign, [(dot, [self, 'make_sound']), (func, [[self],
            (print, ['crying'])])]),
        (assign, [(dot, [self, 'rename']), (func, [[self, name],
            (assign, [(dot, [self, '_name']), name])])]),
        self
    ])
])])
```

これならソースコードとの対応も見やすいと思います。`defclass Animal(name) do ... end` というソースコードは以下のような AST に、

```py
(define, [Animal, (func, [[name], 
    (seq, [
        (define, [self, {}]),
        ...,
        self
    ])
])])
```

`defmethod rename(name) do ... end` というソースコードは以下のような AST に変換してやればよさそうです。

```py
    (assign, [(dot, [self, 'rename']), (func, [[self, name], ...])]),
```

`(define, [self, {}])` などの、どの `defclass` や `defmethod` にも含まれる部分を構文糖衣が自動で埋め込んでやることで、ソースコードから定型部分を消すことができるというわけですね。

では実装にかかりますが、まず `Animal` クラスのガワの部分（メソッドを含まない）の AST を出力するところだけ実装して説明します。`defmethod introduce do ... end` のような `defmethod` や引数のないパターンも実装する必要がありますがやることはほぼ同じなので。

構文糖衣なので `Parser` の変更だけです。

```diff py
 class Parser:
     ...
 
     def _primary(self) -> Expr:
         match self._current_token():
             ...
             case Ident("for"): return self._for()
+            case Ident("defclass"): return self._defclass()
             case Ident(name) if is_ident(name): return self._current_and_advance()
             ...

     ...

     def _for(self):
         ...

+    def _defclass(self):
+        self._current_and_advance()
+        call_expr = self._expression()
+        self._consume(Ident("do"))
+        body_expr = self._expression()
+        self._consume(Ident("end"))
+
+        match call_expr:
+            case (Ident() as ident, params):
+                return (Ident("define"), [ident, (Ident("func"), [params,
+                    (Ident("seq"), [
+                        (Ident("define"), [Ident("self"), {}]),
+                        body_expr,
+                        Ident("self")
+                    ])
+                ])])

     def _binary_left(self, ops, sub_elem):
         ...
```

`defclass` を見つけたら `_defclass()` を呼び出し、クラスの名前と引数（`Animal(name)` に当たる部分。関数呼び出しの形なので `call_expr` としました）およびクラス本体（`body_expr`）を読んでいます。いつものパターンどおりです。

`call_expr` の形によって処理を分けます。今回は関数呼び出しの形（`(<関数>, <引数の配列>)`）を実装しますので、`case (Ident() as ident, params)` で `call_expr` からクラス名（`<関数>` の位置にあるもの）とクラスの引数（`<引数の配列>` の位置にあるもの）を取り出します。

あとは `return` 内で一気に AST を組み上げて返しています。先ほど考えた `defclass` の AST と見比べてください。

これでできているはず。まだ `defmethod` がないのでこじんまりと実行します。

```py
    print(toil.ast(r"""
        defclass Animal(name) do
            self._name = name
        end
    """))
```

出力を整形するとこうなりました。できてますね。

```py
(define, [Animal, (func, [[name],
    (seq, [
        (define, [self, {}]), 
        (assign, [(dot, [self, '_name']), name]),
        self
    ])
])])
```

では残りの部分も実装していきましょう。`inherits` は実装しません。すでに十分目がチカチカしてると思いますので。目がチカチカしなくてすむようにしてからやりましょう。第 4 部までお待ちください。

```diff py
     def _defclass(self):
         self._current_and_advance()
         call_expr = self._expression()
         self._consume(Ident("do"))
         body_expr = self._expression()
         self._consume(Ident("end"))
 
         match call_expr:
             case (Ident() as ident, params):
                 return (Ident("define"), [ident, (Ident("func"), [params,
                     (Ident("seq"), [
                         (Ident("define"), [Ident("self"), {}]),
                         body_expr,
                         Ident("self")
                     ])
                 ])])
+            case Ident() as ident:
+                return (Ident("define"), [ident, (Ident("func"), [[],
+                    (Ident("seq"), [
+                        (Ident("define"), [Ident("self"), {}]),
+                        body_expr,
+                        Ident("self")
+                    ])
+                ])])
+            case _:
+                assert False, f"Invalid defclass syntax @ _defclass(): {call_expr}"
```

クラスの宣言に引数のないパターンと、どちらにも当てはまらないときのエラー処理を追加しました。クラスの宣言に引数がない場合は引数として空の配列を置いています。ほかは同じ。

続いて `_defmethod()` です。

```diff py
 class Parser:
     ...
 
     def _primary(self) -> Expr:
         match self._current_token():
             ...
             case Ident("defclass"): return self._defclass()
+            case Ident("defmethod"): return self._defmethod()
             case Ident(name) if is_ident(name): return self._current_and_advance()
             ...
     def _defclass(self):
         ...

+    def _defmethod(self):
+        self._current_and_advance()
+        call_expr = self._expression()
+        self._consume(Ident("do"))
+        body_expr = self._expression()
+        self._consume(Ident("end"))
+
+        match call_expr:
+            case (Ident() as ident, params):
+                return (Ident("assign"), [
+                    (Ident("dot"), [Ident("self"), ident.name]),
+                    (Ident("func"), [[Ident("self")] + params, body_expr])
+                ])
+            case Ident() as ident:
+                return (Ident("assign"), [
+                    (Ident("dot"), [Ident("self"), ident.name]),
+                    (Ident("func"), [[Ident("self")], body_expr])
+                ])
+            case _:
+                assert False, f"Invalid defmethod syntax @ _defmethod(): {call_expr}"
```

`_defclass()` と似たようなものなのでわかると思います。`dot` の属性名は文字列（`str`）にする仕様なので `ident.name` として文字列を取り出していることと、`[Ident("self")] + params` で引数の配列の先頭に `self` を埋め込んでいるところくらいでしょうか。

実行例です。

```py
    # Example

    print("Defclass and defmethod:")

    print(toil.ast(r"""
        defclass Animal(name) do
            self._name = name;
            defmethod rename(name) do self._name = name end
        end
    """))
    # -> (define, [Animal, (func, [[name], ...])])
    # ->
    # (define, [Animal, (func, [[name],
    #     (seq, [
    #         (define, [self, {}]),
    #         (seq, [
    #             (assign, [(dot, [self, '_name']), name]),
    #             (assign, [(dot, [self, 'rename']), (func, [[self, name],
    #                 (assign, [(dot, [self, '_name']), name])])])
    #         ]),
    #         self
    #     ])
    # ])])
```

2 つめの `# ->` は手で整形した結果です。前回の AST と比較すると内側の `seq` が余分ですが結果には影響ありません。`seq` になっているのは、`body_expr` に当たる部分が `self._name = name; defmethod ... end` と 2 つの式を `;` でつないだ形になっているからです。

AST が想定通り変換されていることを確認してください。

```py
    toil.walk(r"""
        defclass Animal(name) do
            self._name = name;
            defmethod introduce do print("I am", self._name) end;
            defmethod make_sound do print("crying") end;
            defmethod rename(name) do self._name = name end
        end
    """)
    toil.walk(r"""
        animal1 := Animal("Rocky");
        animal2 := Animal("Lucy");
        animal1.introduce();
        animal1.make_sound();
        animal2.introduce();
        animal2.make_sound();
        animal2.rename("Betty");
        animal2.introduce()
    """) # -> I am Rocky\ncrying\nI am Lucy\ncrying\nI am Betty
```

前回と同じ結果になりました。

クラスの宣言に引数がないパターンもやっておきます。

```py
    print(toil.walk(r"""
        defclass Counter do
            self._count = 0;
            defmethod inc(step) do self._count = self._count + step end;
            defmethod get do self._count end
        end;
        c1 := Counter(); c2 := Counter();
        c1.inc(2); c2.inc(5);
        [c1.get(), c2.get()]
    """)) # -> [2, 5]
```

## 第 5 章のまとめ

第 5 章では辞書を実装し、辞書リテラル、ドット記法、UFCS、メソッド記法を追加して最後は構文糖衣でクラス定義を実現しました。

第 5 章には根本的なアイデアは出てきませんでしたが、逆にちょっとした工夫で言語の表現力が広がるという点で面白い章だったと思います。半分くらいは JavaScript パイセンリスペクトっすな章でした[^only-if]。

[^only-if]: JavaScript のオブジェクト指向の実装はプロトタイプ式と言われる方式ですが、ちょっとした（現実にはちょっとしたどころではありませんが）追加でオブジェクトを実装しているのはとても面白いです。クロージャがあってラムダ式も普通に書けるので、組み合わせると応用範囲がすごいです。あとパターンマッチがあって `==` とか `this` とかの変な仕様がなければ気持ちよく使えるんだけどなあ・・・
